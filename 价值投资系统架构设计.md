```python
                    operating_cash_flow=safe_float(latest.get('经营活动产生的现金流量净额')),
                )
            else:  # 港股
                # YFinance获取
                ticker = yf.Ticker(code)
                info = ticker.info
                return FinancialMetrics(
                    code=code,
                    report_date=datetime.now().strftime('%Y-%m-%d'),
                    roe=info.get('returnOnEquity', 0) * 100 if info.get('returnOnEquity') else None,
                    net_margin=info.get('profitMargins', 0) * 100 if info.get('profitMargins') else None,
                    debt_ratio=info.get('debtToEquity', 0) / 100 if info.get('debtToEquity') else None,
                )
        except Exception as e:
            logger.error(f"获取{code}财务数据失败: {e}")
            return FinancialMetrics(code=code, report_date='')

            if code.startswith('6'):

                symbol = f"sz{code}"
            else:
                symbol = code

            if pe_list:
                return {
                    'current': pe_list[-1],
                    'min': min(pe_list),
                    'max': max(pe_list),
                    'median': sorted(pe_list)[len(pe_list)//2],
                    'percentile': sum(1 for x in pe_list if x <= pe_list[-1]) / len(pe_list) * 100
                }
            return {}
        except Exception as e:
            logger.error(f"获取{code}历史PE失败: {e}")
            return {}
    
    def get_risk_free_rate(self) -> float:
        """获取无风险利率（10年期国债收益率）"""
        try:
            df = ak.bond_china_yield(start_date=datetime.now().strftime('%Y%m%d'),
                                    end_date=datetime.now().strftime('%Y%m%d'))
            if not df.empty and '10年期国债收益率' in df.columns:
                return float(df['10年期国债收益率'].iloc[0]) / 100
            return 0.025  # 默认2.5%
        except:
            return 0.025

def safe_float(val, default=None):
    """安全转换浮点数"""
    if val is None or val == '--' or val == 'None':
        return default
    try:
        return float(str(val).replace('%', '').replace('亿', '').replace(',', ''))
    except:
        return default
```

## 三、估值计算层（Valuation Engine）

### 3.1 估值模型实现

```python
# valuation_engine.py
from typing import Dict, Optional, Tuple
from dataclasses import dataclass
import numpy as np

@dataclass
class ValuationResult:
    """估值结果"""
    code: str
    current_price: float
    
    # 各模型估值
    graham_value: Optional[float] = None
    epv_value: Optional[float] = None
    dcf_value: Optional[float] = None
    
    # 估值区间
    fair_value_low: Optional[float] = None   # 保守估值
    fair_value_mid: Optional[float] = None   # 合理估值
    fair_value_high: Optional[float] = None  # 乐观估值
    
    # 安全边际
    margin_of_safety: Optional[float] = None  # 安全边际百分比
    
    # 评分
    valuation_score: int = 0  # 0-100
    
    # 信号
    signal: str = 'HOLD'  # BUY/HOLD/SELL/OBSERVE

class ValuationEngine:
    """估值引擎"""
    
    def __init__(self, risk_free_rate: float = 0.025):
        self.risk_free_rate = risk_free_rate
        self.wacc = 0.08  # 默认折现率8%
        
    def calculate_graham_number(self, eps: float, bvps: float) -> Optional[float]:
        """
        Graham Number 计算
        公式: √(22.5 × EPS × 每股净资产)
        22.5 = PE 15 × PB 1.5 的经验值
        """
        if eps <= 0 or bvps <= 0:
            return None
        return np.sqrt(22.5 * eps * bvps)
    
    def calculate_epv(self, adjusted_earnings: float, 
                      adjusted_assets: float,
                      growth_rate: float = 0.02) -> Optional[float]:
        """
        EPV (Earnings Power Value) 盈利能力价值
        公式: 调整后净利润 / (无风险利率 - 永续增长率)
        
        Args:
            adjusted_earnings: 平均ROE × 调整后净资产
            adjusted_assets: 调整后净资产
            growth_rate: 永续增长率（默认2%）
        """
        if adjusted_earnings <= 0:
            return None
        denominator = self.risk_free_rate - growth_rate
        if denominator <= 0:
            denominator = 0.06  # 保底6%
        return adjusted_earnings / denominator
    
    def calculate_dcf(self, 
                      fcf_history: list,
                      growth_rate_5y: float = 0.10,
                      growth_rate_terminal: float = 0.025,
                      shares_outstanding: float = 1.0) -> Optional[float]:
        """
        DCF (Discounted Cash Flow) 现金流折现
        
        Args:
            fcf_history: 历史自由现金流列表（最近5年）
            growth_rate_5y: 未来5年增长率（默认10%）
            growth_rate_terminal: 永续增长率（默认2.5%）
            shares_outstanding: 总股本（亿股）
        """
        if not fcf_history or len(fcf_history) < 3:
            return None
        
        # 使用平均FCF作为起点
        base_fcf = np.mean(fcf_history[-3:])
        
        # 未来5年FCF预测
        fcf_projected = []
        fcf = base_fcf
        for year in range(1, 6):
            fcf = fcf * (1 + growth_rate_5y)
            fcf_projected.append(fcf)
        
        # 折现
        pv_fcf = sum(fcf / (1 + self.wacc) ** year 
                     for year, fcf in enumerate(fcf_projected, 1))
        
        # 终值计算
        terminal_fcf = fcf_projected[-1] * (1 + growth_rate_terminal)
        terminal_value = terminal_fcf / (self.wacc - growth_rate_terminal)
        pv_terminal = terminal_value / (1 + self.wacc) ** 5
        
        # 企业价值
        enterprise_value = pv_fcf + pv_terminal
        
        # 每股价值（简化处理，假设0净现金）
        per_share_value = enterprise_value / shares_outstanding
        
        return per_share_value
    
    def calculate_historical_percentile(self, current_pe: float, 
                                        pe_history: list) -> float:
        """计算PE历史百分位"""
        if not pe_history or current_pe <= 0:
            return 50.0
        valid_pe = [x for x in pe_history if x > 0]
        if not valid_pe:
            return 50.0
        return sum(1 for x in valid_pe if x <= current_pe) / len(valid_pe) * 100
    
    def comprehensive_valuation(self, 
                                code: str,
                                current_price: float,
                                financial: 'FinancialMetrics',
                                valuation: 'ValuationMetrics',
                                pe_history: list = None) -> ValuationResult:
        """
        综合估值分析
        结合多种估值模型，给出加权估值区间
        """
        result = ValuationResult(code=code, current_price=current_price)
        values = []
        weights = []
        
        # 1. Graham Number（权重30%）
        if financial and valuation:
            eps = current_price / valuation.pe_ttm if valuation.pe_ttm else 0
            bvps = current_price / valuation.pb if valuation.pb else 0
            graham = self.calculate_graham_number(eps, bvps)
            if graham:
                result.graham_value = graham
                values.append(graham)
                weights.append(0.30)
        
        # 2. EPV（权重30%）
        if financial and financial.roe and financial.roe > 0:
            # 简化计算：假设净资产 = 市值/PB
            if valuation and valuation.pb:
                book_value = current_price / valuation.pb
                adjusted_earnings = financial.roe / 100 * book_value
                epv = self.calculate_epv(adjusted_earnings, book_value)
                if epv:
                    result.epv_value = epv
                    values.append(epv)
                    weights.append(0.30)
        
        # 3. 历史PE中位数锚定（权重20%）
        if valuation and valuation.pe_ttm and pe_history:
            median_pe = np.median([x for x in pe_history if x > 0])
            if median_pe > 0:
                eps = current_price / valuation.pe_ttm
                historical_value = eps * median_pe
                values.append(historical_value)
                weights.append(0.20)
        
        # 4. 行业平均PE对比（权重20%，简化处理）
        if valuation and valuation.pe_ttm:
            # 假设行业平均PE为25（可配置）
            industry_avg_pe = 25
            eps = current_price / valuation.pe_ttm
            industry_value = eps * industry_avg_pe
            values.append(industry_value)
            weights.append(0.20)
        
        # 计算加权公允价值区间
        if values and weights:
            # 归一化权重
            total_weight = sum(weights)
            normalized_weights = [w/total_weight for w in weights]
            
            weighted_avg = sum(v * w for v, w in zip(values, normalized_weights))
            
            # 估值区间
            result.fair_value_low = min(values) * 0.9
            result.fair_value_mid = weighted_avg
            result.fair_value_high = max(values) * 1.1
            
            # 安全边际
            if weighted_avg > 0:
                result.margin_of_safety = (weighted_avg - current_price) / weighted_avg * 100
                
                # 估值评分 (0-100)
                if result.margin_of_safety >= 50:
                    result.valuation_score = 90
                elif result.margin_of_safety >= 30:
                    result.valuation_score = 80
                elif result.margin_of_safety >= 15:
                    result.valuation_score = 70
                elif result.margin_of_safety >= 0:
                    result.valuation_score = 60
                elif result.margin_of_safety >= -15:
                    result.valuation_score = 40
                else:
                    result.valuation_score = 20
                
                # 生成信号
                if result.margin_of_safety >= 30 and financial.roe > 15:
                    result.signal = 'BUY'
                elif result.margin_of_safety <= -20 or financial.roe < 5:
                    result.signal = 'SELL'
                elif result.margin_of_safety < 15:
                    result.signal = 'OBSERVE'
                else:
                    result.signal = 'HOLD'
        
        return result
```

## 四、决策引擎层（Decision Engine）

### 4.1 多维度评分体系

```python
# decision_engine.py
from typing import Dict, List
from dataclasses import dataclass
from enum import Enum

class Signal(Enum):
    STRONG_BUY = "强烈买入"
    BUY = "买入"
    HOLD = "持有"
    OBSERVE = "观察"
    REDUCE = "减仓"
    SELL = "卖出"

@dataclass
class QualityMetrics:
    """质量评分维度"""
    profitability_score: int = 0      # 盈利能力 (0-100)
    growth_score: int = 0             # 成长性 (0-100)
    financial_health_score: int = 0   # 财务健康 (0-100)
    stability_score: int = 0          # 稳定性 (0-100)
    moat_score: int = 0               # 护城河 (0-100，定性)

@dataclass
class StockAnalysis:
    """完整股票分析结果"""
    code: str
    name: str
    market: str
    
    # 基础数据
    current_price: float
    market_cap: float
    
    # 估值结果
    valuation: 'ValuationResult'
    
    # 质量评分
    quality: QualityMetrics
    
    # 综合评分
    total_score: int = 0
    ranking: int = 0
    
    # 最终信号
    final_signal: str = 'HOLD'
    confidence: str = '中'  # 高/中/低
    
    # 投资建议
    recommendation: str = ''
    risk_factors: List[str] = None
    opportunity_factors: List[str] = None

class DecisionEngine:
    """决策引擎"""
    
    def __init__(self):
        self.weights = {
            'valuation': 0.35,      # 估值权重35%
            'profitability': 0.25,  # 盈利能力25%
            'growth': 0.15,         # 成长性15%
            'health': 0.15,         # 财务健康15%
            'stability': 0.10,      # 稳定性10%
        }
    
    def calculate_quality_scores(self, financial: 'FinancialMetrics') -> QualityMetrics:
        """计算质量评分"""
        q = QualityMetrics()
        
        # 盈利能力评分
        if financial.roe:
            if financial.roe >= 20:
                q.profitability_score = 100
            elif financial.roe >= 15:
                q.profitability_score = 85
            elif financial.roe >= 10:
                q.profitability_score = 70
            elif financial.roe >= 5:
                q.profitability_score = 50
            else:
                q.profitability_score = 30
        
        # 成长性评分
        if financial.profit_growth:
            if financial.profit_growth >= 30:
                q.growth_score = 100
            elif financial.profit_growth >= 20:
                q.growth_score = 85
            elif financial.profit_growth >= 10:
                q.growth_score = 70
            elif financial.profit_growth >= 0:
                q.growth_score = 50
            else:
                q.growth_score = 30
        
        # 财务健康评分
        health_scores = []
        if financial.debt_ratio:
            if financial.debt_ratio < 40:
                health_scores.append(100)
            elif financial.debt_ratio < 60:
                health_scores.append(80)
            else:
                health_scores.append(50)
        
        if financial.current_ratio:
            if financial.current_ratio > 2:
                health_scores.append(100)
            elif financial.current_ratio > 1.5:
                health_scores.append(80)
            else:
                health_scores.append(60)
        
        if health_scores:
            q.financial_health_score = sum(health_scores) // len(health_scores)
        
        return q
    
    def generate_recommendation(self, analysis: StockAnalysis) -> str:
        """生成投资建议文案"""
        parts = []
        v = analysis.valuation
        q = analysis.quality
        
        # 估值部分
        if v.margin_of_safety and v.margin_of_safety > 0:
            parts.append(f"当前价格低于内在价值{v.margin_of_safety:.1f}%，具备安全边际。")
        elif v.margin_of_safety and v.margin_of_safety < 0:
            parts.append(f"当前价格高于内在价值{abs(v.margin_of_safety):.1f}%，估值偏高。")
        
        # 质量部分
        if q.profitability_score >= 80:
            parts.append("公司盈利能力优秀，ROE保持在较高水平。")
        elif q.profitability_score < 50:
            parts.append("公司盈利能力较弱，需关注改善情况。")
        
        if q.financial_health_score >= 80:
            parts.append("财务状况健康，债务风险可控。")
        
        return " ".join(parts)
    
    def analyze_stock(self, 
                      code: str, 
                      name: str,
                      market: str,
                      price: float,
                      financial: 'FinancialMetrics',
                      valuation_metrics: 'ValuationMetrics',
                      valuation_result: 'ValuationResult') -> StockAnalysis:
        """完整分析一只股票"""
        
        # 计算质量评分
        quality = self.calculate_quality_scores(financial)
        
        # 计算综合评分
        total = (
            valuation_result.valuation_score * self.weights['valuation'] +
            quality.profitability_score * self.weights['profitability'] +
            quality.growth_score * self.weights['growth'] +
            quality.financial_health_score * self.weights['health'] +
            quality.stability_score * self.weights['stability']
        )
        
        # 确定信号
        if valuation_result.signal == 'BUY' and quality.profitability_score >= 70:
            signal = Signal.STRONG_BUY.value
            confidence = '高'
        elif valuation_result.signal == 'BUY':
            signal = Signal.BUY.value
            confidence = '中'
        elif valuation_result.signal == 'SELL' or quality.profitability_score < 40:
            signal = Signal.SELL.value
            confidence = '中'
        elif valuation_result.signal == 'OBSERVE':
            signal = Signal.OBSERVE.value
            confidence = '低'
        else:
            signal = Signal.HOLD.value
            confidence = '中'
        
        analysis = StockAnalysis(
            code=code,
            name=name,
            market=market,
            current_price=price,
            market_cap=valuation_metrics.market_cap if valuation_metrics else 0,
            valuation=valuation_result,
            quality=quality,
            total_score=int(total),
            final_signal=signal,
            confidence=confidence
        )
        
        analysis.recommendation = self.generate_recommendation(analysis)
        
        return analysis
```

```python
        buy_list = [a for a in analyses if a.final_signal in ['强烈买入', '买入']]
        sell_list = [a for a in analyses if a.final_signal == '卖出']

        content.append(f"**买入机会**: {len(buy_list)} 只 | **卖出警示**: {len(sell_list)} 只\n")
        content.append("---\n")
        
        if buy_list:
```

```python
# main.py
import os
import sys
from datetime import datetime
import pandas as pd
from dotenv import load_dotenv

# 加载环境变量
load_dotenv()

class ValueInvestingSystem:
    """价值投资分析系统主控"""
    
    def __init__(self):
        self.collector = DataCollector()
        self.valuation_engine = ValuationEngine()
        self.decision_engine = DecisionEngine()
        self.notifier = FeishuNotifier(os.getenv('FEISHU_WEBHOOK_URL'))
        
        # 股票池（从环境变量或配置文件读取）
        self.stock_pool = os.getenv('STOCK_LIST', '').split(',') if os.getenv('STOCK_LIST') else []
    
    def run_full_scan(self):
        """全市场扫描"""
        print(f"[{datetime.now()}] 开始全市场扫描...")
        
        # 获取A股列表
        stocks = self.collector.get_a_stock_list()
        if stocks.empty:
            print("获取股票列表失败")
            return
        
        # 筛选：市值>50亿，PE>0
        stocks = stocks[(stocks['market_cap'] > 50) & (stocks['pe'] > 0)]
        print(f"待分析股票数: {len(stocks)}")
        
        analyses = []
        for idx, row in stocks.iterrows():
            try:
                code = row['code']
                print(f"分析 {code} {row['name']}...")
                
                # 获取财务数据
                financial = self.collector.get_financial_data(code, 'A股')
                
                # 构建估值指标
                valuation_metrics = ValuationMetrics(
                    code=code,
                    date=datetime.now().strftime('%Y-%m-%d'),
                    price=row['price'],
                    pe_ttm=row['pe'],
                    pb=row['pb'],
                    market_cap=row['market_cap']
                )
                
                # 获取历史PE
                pe_history = self.collector.get_historical_pe(code, years=5)
                
                # 估值计算
                valuation_result = self.valuation_engine.comprehensive_valuation(
                    code=code,
                    current_price=row['price'],
                    financial=financial,
                    valuation=valuation_metrics,
                    pe_history=pe_history.get('history', [])
                )
                
                # 完整分析
                analysis = self.decision_engine.analyze_stock(
                    code=code,
                    name=row['name'],
                    market='A股',
                    price=row['price'],
                    financial=financial,
                    valuation_metrics=valuation_metrics,
                    valuation_result=valuation_result
                )
                
                analyses.append(analysis)
                
            except Exception as e:
                print(f"分析 {row['code']} 失败: {e}")
                continue
        
        # 排序并推送
        analyses.sort(key=lambda x: x.total_score, reverse=True)
        
        # 保存结果
        self.save_results(analyses)
        
        # 推送报告
        self.notifier.send_analysis_report(analyses)
        
        print(f"[{datetime.now()}] 扫描完成，已分析 {len(analyses)} 只股票")
    
    def save_results(self, analyses: List['StockAnalysis']):
        """保存分析结果到CSV"""
        data = []
        for a in analyses:
            data.append({
                'code': a.code,
                'name': a.name,
                'price': a.current_price,
                'signal': a.final_signal,
                'score': a.total_score,
                'margin_of_safety': a.valuation.margin_of_safety,
                'fair_value': a.valuation.fair_value_mid,
                'roe': a.quality.profitability_score,
                'date': datetime.now().strftime('%Y-%m-%d')
            })
        
        df = pd.DataFrame(data)
        filename = f"reports/analysis_{datetime.now().strftime('%Y%m%d')}.csv"
        os.makedirs('reports', exist_ok=True)
        df.to_csv(filename, index=False, encoding='utf-8-sig')
        print(f"结果已保存至 {filename}")

if __name__ == '__main__':
    system = ValueInvestingSystem()
    system.run_full_scan()
```
